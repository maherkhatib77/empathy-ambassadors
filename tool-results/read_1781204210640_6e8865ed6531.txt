    42→    42→

    88→    88→        existing.voted++;

    90→    90→      }

    92→    92→    });

   100→   100→        names: data.names,
   101→   101→      }))

   131→   131→  };

   141→   141→  };

   244→   244→              </CardTitle>
   245→   245→              <p className="text-xs text-muted-foreground print-show">

   248→   248→            </CardHeader>

   251→   251→                <div className="text-center py-12 text-muted-foreground">
