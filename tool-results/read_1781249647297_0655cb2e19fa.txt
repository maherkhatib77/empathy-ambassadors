     1→// ===================================================================
     2→// AdminSettings - ניהול הגדרות (טוגלים, מצב הצבעה, קטגוריות, טקסטים)
     3→// ===================================================================
     4→
     5→'use client';
     6→
     7→import { useState } from 'react';
     8→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     9→import { Button } from '@/components/ui/button';
    10→import { Switch } from '@/components/ui/switch';
    11→import { Input } from '@/components/ui/input';
    12→import { Badge } from '@/components/ui/badge';
    13→import { useAppStore } from '@/store/app-store';
    14→import { DataManager, type VotingMode, DEFAULT_SETTINGS } from '@/lib/data-manager';
    15→import { useTranslation } from '@/lib/translations';
    16→import { ArrowLeft, Settings, Plus, X, Save, RotateCcw, Globe, Type, Vote, Info, Lock, Shield, UserPlus, Trash2 } from 'lucide-react';
    17→
    18→export function AdminSettings() {
    19→  const settings = useAppStore((s) => s.settings);
    20→  const refreshSettings = useAppStore((s) => s.refreshSettings);
    21→  const language = useAppStore((s) => s.language);
    22→  const setCurrentView = useAppStore((s) => s.setCurrentView);
    23→  const { t } = useTranslation();
    24→
    25→  // הגדרות מקומיות לעריכה
    26→  const [votingOpen, setVotingOpen] = useState(settings.voting_open);
    27→  const [submissionOpen, setSubmissionOpen] = useState(settings.submission_open);
    28→  const [votingMode, setVotingMode] = useState<VotingMode>(settings.voting_mode);
    29→
    30→  // קטגוריות
    31→  const [categoriesHe, setCategoriesHe] = useState<string[]>([...settings.categories.he]);
    32→  const [categoriesAr, setCategoriesAr] = useState<string[]>([...settings.categories.ar]);
    33→  const [newCatHe, setNewCatHe] = useState('');
    34→  const [newCatAr, setNewCatAr] = useState('');
    35→
    36→  // טקסטים
    37→  const [textsHe, setTextsHe] = useState<Record<string, string>>({ ...settings.texts.he });
    38→  const [textsAr, setTextsAr] = useState<Record<string, string>>({ ...settings.texts.ar });
    39→
    40→  // ---- סיסמת מנהל ----
    41→  const [currentPass, setCurrentPass] = useState('');
    42→  const [newPass, setNewPass] = useState('');
    43→  const [confirmPass, setConfirmPass] = useState('');
    44→  const [passMsg, setPassMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    45→
    46→  // ---- קודי מנהל משנה ----
    47→  const [adminCodes, setAdminCodes] = useState<string[]>(() => DataManager.getAdminCodes());
    48→  const [newAdminCode, setNewAdminCode] = useState('');
    49→
    50→  // ---- משתמשים מיוחדים ----
    51→  const [customUsers, setCustomUsers] = useState(() => DataManager.getCustomUsers());
    52→  const [newUserId, setNewUserId] = useState('');
    53→  const [newUserName, setNewUserName] = useState('');
    54→  const [newUserClass, setNewUserClass] = useState('');
    55→  const [newUserRole, setNewUserRole] = useState<'student' | 'parent'>('student');
    56→  const [customUserMsg, setCustomUserMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    57→
    58→  // ---- שמירת טוגלים ----
    59→  const handleToggle = (key: 'voting_open' | 'submission_open', value: boolean) => {
    60→    if (key === 'voting_open') setVotingOpen(value);
    61→    else setSubmissionOpen(value);
    62→    DataManager.updateSettings({ [key]: value });
    63→    refreshSettings();
    64→  };
    65→
    66→  // ---- שמירת מצב הצבעה ----
    67→  const handleVotingModeChange = (mode: VotingMode) => {
    68→    setVotingMode(mode);
    69→    DataManager.updateSettings({ voting_mode: mode });
    70→    refreshSettings();
    71→  };
    72→
    73→  // ---- הוספת קטגוריה ----
    74→  const addCategoryHe = () => {
    75→    if (!newCatHe.trim()) return;
    76→    const updated = [...categoriesHe, newCatHe.trim()];
    77→    setCategoriesHe(updated);
    78→    setNewCatHe('');
    79→    saveCategories(updated, categoriesAr);
    80→  };
    81→
    82→  const addCategoryAr = () => {
    83→    if (!newCatAr.trim()) return;
    84→    const updated = [...categoriesAr, newCatAr.trim()];
    85→    setCategoriesAr(updated);
    86→    setNewCatAr('');
    87→    saveCategories(categoriesHe, updated);
    88→  };
    89→
    90→  const removeCategory = (lang: 'he' | 'ar', index: number) => {
    91→    if (lang === 'he') {
    92→      const updated = categoriesHe.filter((_, i) => i !== index);
    93→      setCategoriesHe(updated);
    94→      saveCategories(updated, categoriesAr);
    95→    } else {
    96→      const updated = categoriesAr.filter((_, i) => i !== index);
    97→      setCategoriesAr(updated);
    98→      saveCategories(categoriesHe, updated);
    99→    }
   100→  };
   101→
   102→  const saveCategories = (he: string[], ar: string[]) => {
   103→    DataManager.updateSettings({
   104→      categories: { he, ar },
   105→    });
   106→    refreshSettings();
   107→  };
   108→
   109→  // ---- שמירת טקסטים ----
   110→  const saveTexts = () => {
   111→    DataManager.updateSettings({
   112→      texts: { he: textsHe, ar: textsAr },
   113→    });
   114→    refreshSettings();
   115→  };
   116→
   117→  const updateTextHe = (key: string, value: string) => {
   118→    setTextsHe((prev) => ({ ...prev, [key]: value }));
   119→  };
   120→
   121→  const updateTextAr = (key: string, value: string) => {
   122→    setTextsAr((prev) => ({ ...prev, [key]: value }));
   123→  };
   124→
   125→  // ---- איפוס הגדרות ----
   126→  const resetSettings = () => {
   127→    const msg = language === 'he'
   128→      ? 'האם לאפס את ההגדרות לברירת מחדל?'
   129→      : 'هل تريد إعادة تعيين الإعدادات إلى الافتراضية؟';
   130→    if (window.confirm(msg)) {
   131→      localStorage.removeItem('empathy_settings');
   132→      localStorage.removeItem('empathy_initialized');
   133→      DataManager.initDefaults();
   134→      const fresh = DataManager.getSettings();
   135→      setVotingOpen(fresh.voting_open);
   136→      setSubmissionOpen(fresh.submission_open);
   137→      setVotingMode(fresh.voting_mode);
   138→      setCategoriesHe([...fresh.categories.he]);
   139→      setCategoriesAr([...fresh.categories.ar]);
   140→      setTextsHe({ ...fresh.texts.he });
   141→      setTextsAr({ ...fresh.texts.ar });
   142→      refreshSettings();
   143→    }
   144→  };
   145→
   146→  // ---- שינוי סיסמת מנהל ----
   147→  const handleChangePassword = () => {
   148→    setPassMsg(null);
   149→    if (!currentPass || !newPass || !confirmPass) {
   150→      setPassMsg({ type: 'error', text: t('admin_password_error_empty') });
   151→      return;
   152→    }
   153→    if (currentPass !== DataManager.getAdminPassword()) {
   154→      setPassMsg({ type: 'error', text: t('admin_password_error_wrong') });
   155→      return;
   156→    }
   157→    if (newPass !== confirmPass) {
   158→      setPassMsg({ type: 'error', text: t('admin_password_error_match') });
   159→      return;
   160→    }
   161→    DataManager.setAdminPassword(newPass);
   162→    setPassMsg({ type: 'success', text: t('admin_password_success') });
   163→    setCurrentPass('');
   164→    setNewPass('');
   165→    setConfirmPass('');
   166→  };
   167→
   168→  // ---- ניהול קודי מנהל משנה ----
   169→  const handleAddAdminCode = () => {
   170→    const code = newAdminCode.trim();
   171→    if (!code) return;
   172→    DataManager.addAdminCode(code);
   173→    setAdminCodes(DataManager.getAdminCodes());
   174→    setNewAdminCode('');
   175→  };
   176→
   177→  const handleRemoveAdminCode = (code: string) => {
   178→    DataManager.removeAdminCode(code);
   179→    setAdminCodes(DataManager.getAdminCodes());
   180→  };
   181→
   182→  // ---- ניהול משתמשים מיוחדים ----
   183→  const handleAddCustomUser = () => {
   184→    setCustomUserMsg(null);
   185→    const id = newUserId.trim();
   186→    const name = newUserName.trim();
   187→    if (!id || !name) {
   188→      setCustomUserMsg({ type: 'error', text: language === 'he' ? 'נא למלא תעודת זהות ושם' : 'يرجى ملء رقم الهوية والاسم' });
   189→      return;
   190→    }
   191→    // בדיקה שאין כפילות
   192→    const existing = DataManager.findUserByIdAll(id);
   193→    if (existing) {
   194→      setCustomUserMsg({ type: 'error', text: language === 'he' ? 'תעודת זהות כבר קיימת במערכת' : 'رقم الهوية موجود بالفعل في النظام' });
   195→      return;
   196→    }
   197→    DataManager.addCustomUser({ id, name, role: newUserRole, class: newUserClass.trim() });
   198→    setCustomUsers(DataManager.getCustomUsers());
   199→    setNewUserId('');
   200→    setNewUserName('');
   201→    setNewUserClass('');
   202→    setCustomUserMsg({ type: 'success', text: language === 'he' ? 'המשתמש נוסף בהצלחה' : 'تمت إضافة المستخدم بنجاح' });
   203→    setTimeout(() => setCustomUserMsg(null), 3000);
   204→  };
   205→
   206→  const handleRemoveCustomUser = (userId: string) => {
   207→    DataManager.removeCustomUser(userId);
   208→    setCustomUsers(DataManager.getCustomUsers());
   209→  };
   210→
   211→  const textKeys = Object.keys(textsHe);
   212→
   213→  return (
   214→    <div className="p-4 space-y-6 max-w-4xl mx-auto">
   215→      {/* כותרת */}
   216→      <div className="flex items-center justify-between flex-wrap gap-3">
   217→        <div className="flex items-center gap-3">
   218→          <button
   219→            onClick={() => setCurrentView('admin-dashboard')}
   220→            className="p-2 rounded-full hover:bg-muted transition-colors"
   221→          >
   222→            <ArrowLeft className="w-5 h-5" />
   223→          </button>
   224→          <div>
   225→            <h2 className="text-2xl font-bold flex items-center gap-2">
   226→              <Settings className="w-6 h-6 text-[#0ca7aa]" />
   227→              {language === 'he' ? 'הגדרות' : 'الإعدادات'}
   228→            </h2>
   229→          </div>
   230→        </div>
   231→        <div className="flex gap-2">
   232→          <Button variant="outline" size="sm" onClick={resetSettings}>
   233→            <RotateCcw className="w-4 h-4 ms-1" />
   234→            {language === 'he' ? 'איפוס' : 'إعادة تعيين'}
   235→          </Button>
   236→        </div>
   237→      </div>
   238→
   239→      {/* טוגלים */}
   240→      <Card>
   241→        <CardHeader className="pb-3">
   242→          <CardTitle className="text-lg flex items-center gap-2">
   243→            <Settings className="w-5 h-5" />
   244→            {language === 'he' ? 'מצב מערכת' : 'حالة النظام'}
   245→          </CardTitle>
   246→        </CardHeader>
   247→        <CardContent className="space-y-4">
   248→          {/* הגשה פתוחה */}
   249→          <div className="flex items-center justify-between p-3 rounded-lg bg-muted/30">
   250→            <div>
   251→              <p className="font-medium">
   252→                {language === 'he' ? 'פתח/סגור הגשת פוסטרים' : 'فتح/إغلاق تقديم الملصقات'}
   253→              </p>
   254→              <p className="text-sm text-muted-foreground">
   255→                {language === 'he'
   256→                  ? 'קבע אם תלמידים יכולים להגיש פוסטרים חדשים'
   257→                  : 'حدد ما إذا كان يمكن للطلاب تقديم ملصقات جديدة'}
   258→              </p>
   259→            </div>
   260→            <Switch
   261→              checked={submissionOpen}
   262→              onCheckedChange={(v) => handleToggle('submission_open', v)}
   263→            />
   264→          </div>
   265→
   266→          {/* הצבעה פתוחה */}
   267→          <div className="flex items-center justify-between p-3 rounded-lg bg-muted/30">
   268→            <div>
   269→              <p className="font-medium">
   270→                {language === 'he' ? 'פתח/סגור הצבעה' : 'فتح/إغلاق التصويت'}
   271→              </p>
   272→              <p className="text-sm text-muted-foreground">
   273→                {language === 'he'
   274→                  ? 'קבע אם ההצבעה פתוחה לכולם'
   275→                  : 'حدد ما إذا كان التصويت مفتوحًا للجميع'}
   276→              </p>
   277→            </div>
   278→            <Switch
   279→              checked={votingOpen}
   280→              onCheckedChange={(v) => handleToggle('voting_open', v)}
   281→            />
   282→          </div>
   283→        </CardContent>
   284→      </Card>
   285→
   286→      {/* מצב הצבעה - חשוב! */}
   287→      <Card className="border-[#0ca7aa]/30">
   288→        <CardHeader className="pb-3">
   289→          <CardTitle className="text-lg flex items-center gap-2">
   290→            <Vote className="w-5 h-5 text-[#0ca7aa]" />
   291→            {language === 'he' ? 'מצב הצבעה' : 'وضع التصويت'}
   292→          </CardTitle>
   293→          <CardDescription className="flex items-start gap-2">
   294→            <Info className="w-4 h-4 text-muted-foreground mt-0.5 shrink-0" />
   295→            <span>
   296→              {language === 'he'
   297→                ? 'קבע כיצד המצביעים יכולים להצביע. בחר את המצב המתאים לתחרות.'
   298→                : 'حدد كيف يمكن للمصوتين التصويت. اختر الوضع المناسب للمسابقة.'}
   299→            </span>
   300→          </CardDescription>
   301→        </CardHeader>
   302→        <CardContent className="space-y-3">
   303→          {/* אפשרות 1: הצבעה בודדת */}
   304→          <button
   305→            onClick={() => handleVotingModeChange('single')}
   306→            className={`w-full flex items-start gap-3 p-4 rounded-xl border-2 transition-all text-start ${
   307→              votingMode === 'single'
   308→                ? 'border-[#0ca7aa] bg-[#0ca7aa]/5'
   309→                : 'border-border hover:border-[#0ca7aa]/30'
   310→            }`}
   311→          >
   312→            <div className={`w-5 h-5 rounded-full border-2 mt-0.5 shrink-0 flex items-center justify-center ${
   313→              votingMode === 'single' ? 'border-[#0ca7aa]' : 'border-muted-foreground/30'
   314→            }`}>
   315→              {votingMode === 'single' && <div className="w-2.5 h-2.5 rounded-full bg-[#0ca7aa]" />}
   316→            </div>
   317→            <div>
   318→              <p className="font-semibold flex items-center gap-2">
   319→                <span>1️⃣</span>
   320→                {language === 'he' ? 'הצבעה בודדת' : 'تصويت فردي'}
   321→              </p>
   322→              <p className="text-sm text-muted-foreground mt-1">
   323→                {language === 'he'
   324→                  ? 'כל מצביע יכול לבחור פוסטר אחד בלבד מכל הקטגוריות. הצבעה אחת לכל התחרות.'
   325→                  : 'يمكن لكل مصوت اختيار ملصق واحد فقط من جميع الفئات. تصويت واحد لكل المسابقة.'}
   326→              </p>
   327→            </div>
   328→          </button>
   329→
   330→          {/* אפשרות 2: לפי קטגוריה */}
   331→          <button
   332→            onClick={() => handleVotingModeChange('per_category')}
   333→            className={`w-full flex items-start gap-3 p-4 rounded-xl border-2 transition-all text-start ${
   334→              votingMode === 'per_category'
   335→                ? 'border-[#0ca7aa] bg-[#0ca7aa]/5'
   336→                : 'border-border hover:border-[#0ca7aa]/30'
   337→            }`}
   338→          >
   339→            <div className={`w-5 h-5 rounded-full border-2 mt-0.5 shrink-0 flex items-center justify-center ${
   340→              votingMode === 'per_category' ? 'border-[#0ca7aa]' : 'border-muted-foreground/30'
   341→            }`}>
   342→              {votingMode === 'per_category' && <div className="w-2.5 h-2.5 rounded-full bg-[#0ca7aa]" />}
   343→            </div>
   344→            <div>
   345→              <p className="font-semibold flex items-center gap-2">
   346→                <span>📑</span>
   347→                {language === 'he' ? 'הצבעה לפי קטגוריה' : 'تصويت حسب الفئة'}
   348→              </p>
   349→              <p className="text-sm text-muted-foreground mt-1">
   350→                {language === 'he'
   351→                  ? 'כל מצביע יכול להצביע פעם אחת בכל קטגוריה. לדוגמה: הצבעה אחת ב"סובלנות" והצבעה אחת ב"איכות הסביבה".'
   352→                  : 'يمكن لكل مصوت التصويت مرة واحدة في كل فئة. مثال: تصويت واحد في "التسامح" وتصويت واحد في "جودة البيئة".'}
   353→              </p>
   354→            </div>
   355→          </button>
   356→
   357→          {/* מצב נוכחי */}
   358→          <div className="flex items-center gap-2 pt-2">
   359→            <Badge className={votingMode === 'single' ? 'bg-purple-500' : 'bg-[#0ca7aa]'} variant="secondary">
   360→              {votingMode === 'single'
   361→                ? (language === 'he' ? '✓ הצבעה בודדת' : '✓ تصويت فردي')
   362→                : (language === 'he' ? '✓ לפי קטגוריה' : '✓ حسب الفئة')}
   363→            </Badge>
   364→          </div>
   365→        </CardContent>
   366→      </Card>
   367→
   368→      {/* קטגוריות */}
   369→      <Card>
   370→        <CardHeader className="pb-3">
   371→          <CardTitle className="text-lg flex items-center gap-2">
   372→            <Globe className="w-5 h-5" />
   373→            {language === 'he' ? 'קטגוריות' : 'الفئات'}
   374→          </CardTitle>
   375→        </CardHeader>
   376→        <CardContent className="space-y-6">
   377→          {/* קטגוריות בעברית */}
   378→          <div>
   379→            <p className="font-medium mb-2 text-sm">
   380→              {language === 'he' ? 'קטגוריות בעברית' : 'الفئات بالعبرية'}
   381→            </p>
   382→            <div className="flex flex-wrap gap-2 mb-2">
   383→              {categoriesHe.map((cat, idx) => (
   384→                <span
   385→                  key={idx}
   386→                  className="inline-flex items-center gap-1 px-3 py-1 rounded-full bg-[#0ca7aa]/10 text-[#0ca7aa] text-sm"
   387→                >
   388→                  {cat}
   389→                  <button
   390→                    onClick={() => removeCategory('he', idx)}
   391→                    className="hover:text-destructive transition-colors"
   392→                  >
   393→                    <X className="w-3 h-3" />
   394→                  </button>
   395→                </span>
   396→              ))}
   397→            </div>
   398→            <div className="flex gap-2">
   399→              <Input
   400→                value={newCatHe}
   401→                onChange={(e) => setNewCatHe(e.target.value)}
   402→                onKeyDown={(e) => e.key === 'Enter' && addCategoryHe()}
   403→                placeholder={language === 'he' ? 'קטגוריה חדשה...' : 'فئة جديدة...'}
   404→                className="flex-1"
   405→              />
   406→              <Button size="sm" onClick={addCategoryHe} variant="outline">
   407→                <Plus className="w-4 h-4" />
   408→              </Button>
   409→            </div>
   410→          </div>
   411→
   412→          {/* קטגוריות בערבית */}
   413→          <div>
   414→            <p className="font-medium mb-2 text-sm">
   415→              {language === 'he' ? 'קטגוריות בערבית' : 'الفئات بالعربية'}
   416→            </p>
   417→            <div className="flex flex-wrap gap-2 mb-2">
   418→              {categoriesAr.map((cat, idx) => (
   419→                <span
   420→                  key={idx}
   421→                  className="inline-flex items-center gap-1 px-3 py-1 rounded-full bg-[#0ca7aa]/10 text-[#0ca7aa] text-sm"
   422→                >
   423→                  {cat}
   424→                  <button
   425→                    onClick={() => removeCategory('ar', idx)}
   426→                    className="hover:text-destructive transition-colors"
   427→                  >
   428→                    <X className="w-3 h-3" />
   429→                  </button>
   430→                </span>
   431→              ))}
   432→            </div>
   433→            <div className="flex gap-2">
   434→              <Input
   435→                value={newCatAr}
   436→                onChange={(e) => setNewCatAr(e.target.value)}
   437→                onKeyDown={(e) => e.key === 'Enter' && addCategoryAr()}
   438→                placeholder={language === 'he' ? 'קטגוריה חדשה...' : 'فئة جديدة...'}
   439→                className="flex-1"
   440→              />
   441→              <Button size="sm" onClick={addCategoryAr} variant="outline">
   442→                <Plus className="w-4 h-4" />
   443→              </Button>
   444→            </div>
   445→          </div>
   446→        </CardContent>
   447→      </Card>
   448→
   449→      {/* שינוי סיסמת מנהל */}
   450→      <Card className="border-[#e74c3c]/30">
   451→        <CardHeader className="pb-3">
   452→          <CardTitle className="text-lg flex items-center gap-2">
   453→            <Lock className="w-5 h-5 text-[#e74c3c]" />
   454→            {t('admin_password_title')}
   455→          </CardTitle>
   456→        </CardHeader>
   457→        <CardContent className="space-y-3">
   458→          <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
